//@Preview(showBackground = true)
@Composable
private fun ListItem() {
    var counter by remember { mutableIntStateOf(0) }//связал перем с состоянием
    Card(
        modifier = Modifier
            .fillMaxWidth()
            .height(250.dp)
            .padding(25.dp)
            .clickable {// сдесь фу поведения card при клике увеличив перем на ед
                counter++//подключил переменную в виде счетчика
                Log.d("MY_TEG", "Card clicked:$counter")// запустил logcat

            },
        shape = RoundedCornerShape(15.dp),
        elevation = CardDefaults.cardElevation(
            defaultElevation = 8.dp
        )
    )

    {
        Box(
            modifier = Modifier
                .background(Color.Blue)
                .fillMaxWidth(),
            contentAlignment = Alignment.Center

        )
        {
            Row(
                modifier = Modifier
                    .background(Color.Yellow),
                verticalAlignment = Alignment.CenterVertically,
            ) {
                Image(
                    painter = painterResource(id = R.drawable.tranzist),
                    contentDescription = "my image",
                    contentScale = ContentScale.Crop,
                    modifier = Modifier
                        .padding(5.dp)
                        .size(150.dp)
                        .clip(CircleShape)

                )

                Text(text = "$counter",// здесь выводим переменную $counter

                        fontSize = 125.sp)//сдесь размер букв

            }
        }
    }
}

class MainActivity : ComponentActivity() {

        super.onCreate(savedInstanceState)

        setContent {
            LazyColumn(
                modifier = Modifier
                    .fillMaxSize()
                    .padding(10.dp)
                    .background(Color.Magenta),
                horizontalAlignment = Alignment.CenterHorizontally,
                verticalArrangement = Arrangement.Center,

                )
            {
                items(50) { index ->// количество элементов
                    Text(
                        text = "my text ${index + 1}",//переменная
                        style = TextStyle
                            (fontSize = 45.sp),
                        color = Color.White
                    )
                }
            }
        }
    }
}

/*
                    ОБЯСНЕНИЕ LazyColume и items
     Функция items работает как умный конвейер:
     • Рисует только видимое: LazyColumn смотрит на размер экрана телефона и
      просит блок items создать только первые элементы Text,
     которые помещаются на экран прямо сейчас.
     Остальные элементы даже не создаются в памяти.
     • Переиспользует элементы при прокрутке: Когда вы листаете список вниз,
     верхний Text уходит за пределы экрана и уничтожается,
     а items тут же создает новый Text снизу.
     • Понимает структуру списка: Сам LazyColumn — это просто пустая коробка для прокрутки,
      которая принимает только «инструкции» о том, какие блоки в ней будут лежать.
     Блок item говорит: «Вот инструкция для одного блока (например, заголовка)»,
      а items говорит: «Вот шаблон, по которому нужно штамповать много одинаковых блоков».

*/

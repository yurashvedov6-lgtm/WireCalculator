//@Preview(showBackground = true)
@Composable
private fun ListItem(name: String, data: String) {
    Card(
        modifier = Modifier
            .fillMaxWidth()
            .padding(25.dp),
            //.clickable{
               // Log.d("MY_TAG", "Card clicked!")
            //}



           // .offset(x = 25.dp),// отступ от оригинального положения
        shape = RoundedCornerShape(15.dp),
        elevation = CardDefaults.cardElevation(
            defaultElevation = 8.dp
        )
    )

    {
        Box(
            modifier = Modifier
                .background(Color.Blue)
                .fillMaxWidth()
                .pointerInput(Unit) {

                    detectDragGesturesAfterLongPress(
                        onDragStart = { offset ->
                            Log.d("MY_TEG", "Старт")
                        },
                        // Явно пишем имя параметра onDrag:
                        onDrag = { change, dragAmount ->
                            change.consume()
                            Log.d("MY_TEG", "Движение: $dragAmount")
                        }
                    )
                },
            contentAlignment = Alignment.Center

        )
        {
            Row(
                modifier = Modifier
                    .background(Color.Yellow),
                verticalAlignment = Alignment.CenterVertically,
            ) {
                Image(
                    painter = painterResource(id = R.drawable.tranzist),
                    contentDescription = "my image",
                    contentScale = ContentScale.Crop,
                    modifier = Modifier
                        .padding(5.dp)
                        .size(64.dp)
                        .clip(CircleShape)

                )

                Text(text = " behavior and decoration ")
            }
        }
    }
}


//описание кода первое изучаем Modifier  он имеет две состовляющие декаративный
//это .filMaxSize , .background
// и вторая состовляющая это behavior поведенческая тоесть какието действвия
// это у нас clickable() и pointerInput (Unit)
//все эти действия поведения элементов image, row мы проверяем в logcat
//такой строкой  Log.d("MY_TEG", "Старт") и то что в скобаках первый ковычки
// это фильтр мы его записываем в logcat в фильтр по нему и будет показывать нам
// следующие ковычки это то какие действия происходят

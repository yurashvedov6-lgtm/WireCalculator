@Preview(showBackground = true)
@Composable
fun NewText() {
    Row(
        modifier = Modifier
            .background(Color.Gray)
            .fillMaxSize(),
        horizontalArrangement = Arrangement.SpaceEvenly
    )
    {
        Column(
            modifier = Modifier
                .background(Color.Yellow)
                .fillMaxHeight(0.5f)
                .fillMaxWidth(0.5f),
            verticalArrangement = Arrangement.SpaceEvenly,
            horizontalAlignment = Alignment.CenterHorizontally
        ) {
            Text(text = "My first ")
            Text(text = "My first ")
            Text(text = "My first ")
        }
        Column(
            modifier = Modifier
                .background(Color.Blue)
                .fillMaxHeight(0.5f)
                .fillMaxWidth(),
            verticalArrangement = Arrangement.SpaceEvenly,
            horizontalAlignment = Alignment.CenterHorizontally
        ) {
            Text(text = "My first ")
            Text(text = "My first ")
            Text(text = "My first ")
        }
    }
}

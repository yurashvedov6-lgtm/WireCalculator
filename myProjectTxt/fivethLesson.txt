//@Preview(showBackground = true)
@Composable
private fun StateBox() {
    var counter by remember { mutableIntStateOf(0) }
    var boxColor by remember { mutableStateOf(Color.LightGray) }
    Box(
        modifier = Modifier
            .padding(5.dp)
            .size(245.dp)
            .background(
                boxColor,
                shape = CircleShape
            )
            .clickable {
                when (++counter) {
                    5 -> boxColor = Color.Blue
                    10 ->boxColor = Color.Magenta
                }
                Log.d("MY_TAG", "my counter:$counter")
            },
        contentAlignment = Alignment.Center


    )
    {
        Text(text = "myBox $counter",
            style = TextStyle
        (fontSize = 45.sp, color = Color.White))
    }
}

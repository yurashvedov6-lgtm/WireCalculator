class MainActivity : ComponentActivity() {

        super.onCreate(savedInstanceState)

        // Разворачиваем наш UI внутри MainActivity
        setContent {
            Column(
                modifier = Modifier
                    .fillMaxSize()
                    .padding(5.dp)
                    .background(Color.Magenta, shape = CircleShape),

                horizontalAlignment = Alignment.CenterHorizontally,
                verticalArrangement = Arrangement.Center
            )
            {
                StateBox()
            }

        }
    }
}


//@Preview(showBackground = true)
@Composable
private fun StateBox() {
    var counter by remember { mutableIntStateOf(0) }
    //var boxColor by remember { mutableStateOf(Color.Green) }
    Box(
        modifier = Modifier
            .size(100.dp)
            /*внизу первый рабочий самый продвинутый код*/
            .background(
                color = when {
                    counter >= 10 -> Color.Blue
                    counter >= 5 -> Color.Yellow
                    else -> Color.Green
                },
                shape = CircleShape
            )
            .clickable {
                counter++
                Log.d("MY_TAG", "счет идет :$counter ")
            },
        contentAlignment = Alignment.Center
    )

    {
        Text(
            text = "$counter ",
            style = TextStyle(Color.Magenta, fontSize = 45.sp)
        )
    }
}

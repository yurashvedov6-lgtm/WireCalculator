//@Preview(showBackground = true)
@Composable
private fun StateButton() {
    var counter by remember { mutableIntStateOf(0) }
    var boxColor by remember { mutableStateOf(Color.LightGray) }
    Box(
        modifier = Modifier
            .size(100.dp)
            .background(boxColor, shape = CircleShape)
            .clickable{
                counter++
                Log.d("MY_TAG" , "счет идет :$counter ")
                boxColor = when {
                    counter >= 10 -> Color.Green
                    counter >= 5 -> Color.Red
                    else -> Color.LightGray
                }
            },
        contentAlignment = Alignment.Center,



        ) {
        Text(
            text = "$counter ",
            color = Color.Yellow,
            fontSize = 45.sp
        )
    }
}
